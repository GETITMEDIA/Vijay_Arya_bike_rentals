----------------------------------------------------------------
1) FIRESTORE RULES
   Firebase console → Build → Firestore Database → Rules
----------------------------------------------------------------

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // Vehicles: everyone can read (the website shows them),
    // only a signed-in admin can add / edit / delete.
    match /vehicles/{vehicleId} {
      allow read: if true;
      allow write: if request.auth != null;
    }

    // Bookings: a customer may create their own booking,
    // only a signed-in admin can read, edit or delete them.
    match /bookings/{bookingId} {
      allow create: if request.resource.data.bookingId is string
                    && request.resource.data.customerName is string
                    && request.resource.data.customerPhone is string;
      allow read, update, delete: if request.auth != null;
    }

    // Nothing else is reachable
    match /{document=**} {
      allow read, write: if false;
    }
  }
}


----------------------------------------------------------------
2) STORAGE RULES
   Firebase console → Build → Storage → Rules
----------------------------------------------------------------

rules_version = '2';
service firebase.storage {
  match /b/{bucket}/o {

    // Vehicle photos: public to read, admin-only to change
    match /vehicles/{vehicleId}/{fileName} {
      allow read: if true;
      allow write: if request.auth != null
                   && request.resource.size < 5 * 1024 * 1024
                   && request.resource.contentType.matches('image/.*');
    }

    // KYC documents: a customer may upload during booking,
    // ONLY a signed-in admin can view or download them.
    match /kyc/{bookingId}/{fileName} {
      allow read: if request.auth != null;
      allow create: if request.resource.size < 8 * 1024 * 1024
                    && (request.resource.contentType.matches('image/.*')
                        || request.resource.contentType == 'application/pdf');
      allow update, delete: if request.auth != null;
    }

    match /{allPaths=**} {
      allow read, write: if false;
    }
  }
}

3) ADMIN LOGIN
   Firebase console → Build → Authentication → Sign-in method
   → Email/Password → Enable → Save

   Then: Users tab → Add user
     Email:    (the shop's admin email)
     Password: (a strong password)

   That email + password becomes the admin panel login.
   Add more staff logins the same way.
